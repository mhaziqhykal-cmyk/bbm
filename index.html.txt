        // Tab switching logic
        function switchTab(tabId) {
            // Stop speech when switching tab
            stopAudio();

            // Hide all tab content
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(content => content.classList.add('hidden'));

            // Remove active style from all tab buttons
            const buttons = document.querySelectorAll('.tab-btn');
            buttons.forEach(btn => btn.classList.remove('active'));

            // Show selected tab content
            document.getElementById(`content-${tabId}`).classList.remove('hidden');

            // Highlight selected button
            document.getElementById(`tab-${tabId}`).classList.add('active');
        }

        // ================= SPEECH SYNTHESIS / AUDIO PLAYER LOGIC =================
        const dialogData = [
            "Amir berkata: Kawan-kawan, cikgu kata kumpulan kita bertanggungjawab untuk merancang Projek Kebun Sains Sekolah minggu depan. Apa cadangan tanaman yang patut kita tanam?",
            "Bella berkata: Saya cadangkan kita tanam sayuran hijau seperti bayam dan kangkung. Sayuran ini cepat tumbuh dan mudah dijaga. Kita cuma perlu siram dua kali sehari.",
            "Chentan berkata: Cadangan yang bagus, Bella! Tapi bagaimana dengan pokok herba seperti pokok lidah buaya dan serai? Herba sangat berguna untuk kegunaan kesihatan dan tidak mudah diserang serangga perosak.",
            "Dina berkata: Saya setuju dengan kedua-dua cadangan! Kita boleh bahagikan kebun kepada dua kawasan: satu bahagian untuk sayuran hijau dan satu bahagian lagi untuk pokok herba. Selain itu, kita juga perlu sediakan jadual bertugas untuk menyiram dan memupuk baja.",
            "Amir berkata: Wah, idea yang sangat bernas, Dina! Jadi keputusan kumpulan kita: kita akan tanam sayur hijau dan herba, serta membuat jadual bertugas harian. Mari kita catat isi ini untuk dibentangkan kepada cikgu."
        ];

        let currentSentenceIndex = 0;
        let isSpeaking = false;
        let isPaused = false;
        let synth = window.speechSynthesis;

        function playAudio() {
            if (!('speechSynthesis' in window)) {
                alert("Minta maaf, pelayar web anda tidak menyokong fungsi audio penceritaan.");
                return;
            }

            if (isPaused) {
                synth.resume();
                isPaused = false;
                isSpeaking = true;
                updateAudioStatus("▶️ Menyambung bacaan audio...");
                return;
            }

            if (isSpeaking) return;

            synth.cancel(); // Reset previous
            speakSentence(currentSentenceIndex);
        }

        function speakSentence(index) {
            if (index >= dialogData.length) {
                stopAudio();
                updateAudioStatus("✅ Bacaan audio selesai. Sila teruskan ke Stesen Perbincangan!");
                return;
            }

            currentSentenceIndex = index;
            highlightDialog(index);

            const utterance = new SpeechSynthesisUtterance(dialogData[index]);
            utterance.lang = 'ms-MY'; // Malay voice if available, fallback default
            utterance.rate = parseFloat(document.getElementById('rate').value) || 0.9;
            utterance.pitch = parseFloat(document.getElementById('pitch').value) || 1.1;

            utterance.onstart = function() {
                isSpeaking = true;
                updateAudioStatus(`🔊 Membaca Bahagian ${index + 1} daripada ${dialogData.length}...`);
            };

            utterance.onend = function() {
                if (isSpeaking && !isPaused) {
                    currentSentenceIndex++;
                    speakSentence(currentSentenceIndex);
                }
            };

            utterance.onerror = function() {
                stopAudio();
                updateAudioStatus("⚠️ Ralat main semula audio.");
            };

            synth.speak(utterance);
        }

        function pauseAudio() {
            if (synth.speaking && !synth.paused) {
                synth.pause();
                isPaused = true;
                isSpeaking = false;
                updateAudioStatus("⏸️ Audio dijeda.");
            }
        }

        function stopAudio() {
            synth.cancel();
            isSpeaking = false;
            isPaused = false;
            currentSentenceIndex = 0;
            removeHighlights();
            updateAudioStatus("⏹️ Audio dihentikan. Tekan 'Main Audio' untuk mula semula.");
        }

        function highlightDialog(index) {
            removeHighlights();
            const el = document.getElementById(`dialog-${index}`);
            if (el) {
                el.classList.add('bg-yellow-200', 'ring-4', 'ring-cartoon-pink', 'scale-[1.01]');
                el.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            }
        }

        function removeHighlights() {
            const items = document.querySelectorAll('.dialog-item');
            items.forEach(item => {
                item.classList.remove('bg-yellow-200', 'ring-4', 'ring-cartoon-pink', 'scale-[1.01]');
            });
        }

        function updateAudioStatus(msg) {
            const statusEl = document.getElementById('audio-status');
            if (statusEl) statusEl.innerText = msg;
        }

        // Station switching logic
        function showStation(stnNum) {
            // Hide all station contents
            const stnContents = document.querySelectorAll('.stn-content');
            stnContents.forEach(c => c.classList.add('hidden'));

            // Reset buttons style
            const stnBtns = document.querySelectorAll('.stn-btn');
            stnBtns.forEach(btn => {
                btn.classList.remove('bg-cartoon-blue', 'text-white');
                btn.classList.add('bg-white', 'text-cartoon-dark');
            });

            // Show current station content
            document.getElementById(`stn-content-${stnNum}`).classList.remove('hidden');

            // Set current button active
            const activeBtn = document.getElementById(`stn-btn-${stnNum}`);
            activeBtn.classList.remove('bg-white', 'text-cartoon-dark');
            activeBtn.classList.add('bg-cartoon-blue', 'text-white');

            // Change box background color based on station
            const box = document.getElementById('station-box');
            if (stnNum === 1) box.className = "bg-blue-50 border-pop p-6 rounded-3xl relative transition-all";
            if (stnNum === 2) box.className = "bg-yellow-50 border-pop p-6 rounded-3xl relative transition-all";
            if (stnNum === 3) box.className = "bg-pink-50 border-pop p-6 rounded-3xl relative transition-all";
        }

        // Quiz checking logic
        function checkQuiz(e) {
            e.preventDefault();
            const q1 = document.querySelector('input[name="q1"]:checked')?.value;
            const q2 = document.querySelector('input[name="q2"]:checked')?.value;
            const q3 = document.querySelector('input[name="q3"]:checked')?.value;

            const resultDiv = document.getElementById('quiz-result');
            resultDiv.classList.remove('hidden');

            if (!q1 || !q2 || !q3) {
                resultDiv.className = "mt-4 p-4 rounded-2xl border-pop text-center font-bold bg-cartoon-orange text-white";
                resultDiv.innerHTML = "<i class='fa-solid fa-triangle-exclamation mr-1'></i> Sila jawab SEMUA soalan kuiz sebelum menyemak!";
                return;
            }

            let score = 0;
            if (q1 === 'B') score++;
            if (q2 === 'B') score++;
            if (q3 === 'A') score++;

            if (score === 3) {
                resultDiv.className = "mt-4 p-4 rounded-2xl border-pop text-center font-bold bg-cartoon-green text-white";
                resultDiv.innerHTML = "🎉 SYABAS! Kumpulan anda menjawab 3/3 Soalan dengan BETUL! Kemahiran mendengar anda sangat cemerlang!";
            } else {
                resultDiv.className = "mt-4 p-4 rounded-2xl border-pop text-center font-bold bg-cartoon-yellow text-cartoon-dark";
                resultDiv.innerHTML = `👍 Boleh tahan! Kumpulan anda mendapat ${score}/3 betul. Sila baca/dengar semula skrip dialog untuk jawapan tepat.`;
            }
        }

        // Save notes to localStorage
        function saveNotes() {
            const kumpulan = document.getElementById('input-kumpulan').value;
            const ringkasan = document.getElementById('input-ringkasan').value;
            const refleksi = document.getElementById('input-refleksi').value;

            const data = { kumpulan, ringkasan, refleksi };
            localStorage.setItem('m11_pdp_notes', JSON.stringify(data));

            const toast = document.getElementById('toast-msg');
            toast.classList.remove('hidden');
            setTimeout(() => {
                toast.classList.add('hidden');
            }, 3000);
        }

        // Load saved notes on startup
        window.onload = function() {
            const saved = localStorage.getItem('m11_pdp_notes');
            if (saved) {
                const data = JSON.parse(saved);
                if (data.kumpulan) document.getElementById('input-kumpulan').value = data.kumpulan;
                if (data.ringkasan) document.getElementById('input-ringkasan').value = data.ringkasan;
                if (data.refleksi) document.getElementById('input-refleksi').value = data.refleksi;
            }
        };
